# runCheck (serveurs locaux ; latence retirée)
http 200 → {"code":200,"state":"ok"}
http 200 + mot-clé présent → {"code":200,"state":"ok"}
http 200 + mot-clé absent → {"code":200,"error":"mot-clé « absent » absent","state":"crit"}
http 500 → {"code":500,"error":"HTTP 500","state":"crit"}
http 404 sans expect → {"code":404,"state":"ok"}
http 404 attendu → {"code":404,"state":"ok"}
http 404, attendus [200, 204] → {"code":404,"error":"HTTP 404","state":"crit"}
http 302 non suivie → {"code":302,"state":"ok"}
http délai → {"code":0,"error":"timeout","state":"crit"}
http port fermé → {"code":0,"error":"ECONNREFUSED","state":"crit"}
tcp ouvert → {"code":"tcp","state":"ok"}
tcp fermé → {"code":0,"error":"ECONNREFUSED","state":"crit"}
dns deux adresses → {"code":"dns","state":"ok"}
dns adresse attendue présente → {"code":"dns","state":"ok"}
dns adresse attendue absente → {"code":"dns","error":"réponse 192.0.2.7, 192.0.2.8, attendu 192.0.2.9","state":"crit"}
dns aucune adresse → {"code":0,"error":"ENODATA","state":"crit"}
tls J-10 (défauts 21/7) → {"code":"tls","state":"warn","tls":{"daysLeft":10,"issuer":"vigie tests","validTo":"date"}}
tls J-10, warnDays 7 → {"code":"tls","state":"ok","tls":{"daysLeft":10,"issuer":"vigie tests","validTo":"date"}}
tls J-10, critDays 14 → {"code":"tls","state":"crit","tls":{"daysLeft":10,"issuer":"vigie tests","validTo":"date"}}
tls expiré → {"code":"tls","error":"certificat expiré","state":"crit","tls":{"daysLeft":-2,"issuer":"vigie tests","validTo":"date"}}
tls port fermé → {"code":0,"error":"ECONNREFUSED","state":"crit"}
